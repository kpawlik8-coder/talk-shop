%import common.WS
%ignore WS

start: price_question | availability_question | quantity_question

price_question: HOW_MUCH BE DET? ITEM

availability_question: BE DET? ITEM IN_STOCK
                     | DO_YOU_HAVE DET? ITEM

quantity_question: HOW_MANY ITEM BE IN_STOCK
                 | HOW_MANY ITEM DO_YOU_HAVE

HOW_MUCH: "how much"
HOW_MANY: "how many"
BE: "is" | "are"
DET: "the" | "a" | "any"
IN_STOCK: "in stock"
DO_YOU_HAVE: "do you have"
ITEM: /[a-z]+(_[a-z]+)*/
